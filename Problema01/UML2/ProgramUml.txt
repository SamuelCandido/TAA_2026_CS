skinparam classAttributeIconSize 0

package Problema01 {
    enum UnidadeDePeso {
        GRAMA = 1
        QUILOGRAMA = 1000
        TONELADA = 1000000
    }

    enum TipoDeEntrega {
        ENCOMENDA_PAC
        RETIRADA_LOCAL
        SEDEX
    }

    class TipoDeEntregaFactory {
        + {static} CriarTipoDeEntrega(tipoEntidade: TipoDeEntrega): ITipoDeEntrega
    }
}

package Problema01.TiposDeEntrega {
    struct Peso <<Value Object>> {
        + EmGramas: decimal
        + EmQuilogramas: decimal
        + EmToneladas: decimal
        + Peso(valor: decimal, unidade: UnidadeDePeso)
    }

    interface ITipoDeEntrega {
        + CalcularValorEntrega(peso: Peso): decimal
    }

    class RetiradaLocal {
        - {static} ValorRetiradaLocal: decimal
        + CalcularValorEntrega(peso: Peso): decimal
    }

    class EncomendaPAC {
        - {static} PesoMaximoKg: decimal
        - {static} PesoLimiteLeveKg: decimal
        - {static} ValorPesoLeve: decimal
        - {static} ValorPesoPesado: decimal
        + CalcularValorEntrega(peso: Peso): decimal
    }

    class Sedex {
        - {static} PesoLimiteMinimoKg: decimal
        - {static} PesoLimiteLeveKg: decimal
        - {static} ValorPesoMinimo: decimal
        - {static} ValorPesoLeve: decimal
        - {static} ValorBase: decimal
        - {static} ValorPorBlocoAdicional: decimal
        - {static} GramasPorBloco: decimal
        + CalcularValorEntrega(peso: Peso): decimal
    }
}

RetiradaLocal ..|> ITipoDeEntrega
EncomendaPAC ..|> ITipoDeEntrega
Sedex ..|> ITipoDeEntrega

TipoDeEntregaFactory ..> ITipoDeEntrega : <<cria>>
TipoDeEntregaFactory ..> TipoDeEntrega : <<usa>>

ITipoDeEntrega ..> Peso : <<usa>>
Peso ..> UnidadeDePeso : <<usa>>
@enduml
