' ENUMS

package "Enums" {
    enum ModoCasaEnum {
        Sono
        Trabalho
    }

    enum TipoLampadaEnum {
        Phellipes
        Shoyumi
    }

    enum TipoArCondicionadoEnum {
        Ventobaumn
        Gellakaza
    }

    enum TipoPersianaEnum {
        Solarius
        Natlight
    }
}

' =====================================================
' INTERFACES (PADRÃO DO SISTEMA)
' =====================================================
package "Interfaces" {

    interface ILampada {
        + Ligar()
        + Desligar()
        + EstaLigada(): bool
    }

    interface IPersiana {
        + SubirPersiana()
        + DescerPersiana()
        + EstaAberta(): bool
    }

    interface IArCondicionado {
        + Ligar()
        + Desligar()
        + AumentarTemperatura()
        + DiminuirTemperatura()
        + DefinirTemperatura(int)
        + EstaLigado(): bool
        + GetTemperatura(): int
    }

    interface IDispositivoObserver {
        + Atualizar(modo: ModoCasaEnum)
    }
}
